#+title: Noonker's Config
#+property: header-args:emacs-lisp :tangle ~/.emacs

A literate Emacs config.  =make home= tangles it into =~/.emacs= at build time: see =emacs-init-file= in =guix/channels/noonker/home/emacs.scm=.

Packages come from Guix, in the Emacs section of =guix/home-configuration.scm= and in =guix/channels/noonker/packages/emacs.scm=.  =use-package= here only controls when they load and how they're configured.

#+begin_src emacs-lisp
(defun uv-activate ()
  "Activate Python environment managed by uv based on current project directory.
Looks for .venv directory in project root and activates the Python interpreter."
  (interactive)
  (let* ((project-root (project-root (project-current t)))
         (venv-path (expand-file-name ".venv" project-root))
         (python-path (expand-file-name
                       "bin/python"
                       venv-path)))
    (if (file-exists-p python-path)
        (progn
          ;; Set Python interpreter path
          (setq python-shell-interpreter python-path)

          ;; Update exec-path to include the venv's bin directory
          (let ((venv-bin-dir (file-name-directory python-path)))
            (setq exec-path (cons venv-bin-dir
                                  (remove venv-bin-dir exec-path))))

          ;; Update PATH environment variable
          (setenv "PATH" (concat (file-name-directory python-path)
                                 path-separator
                                 (getenv "PATH")))

          ;; Update VIRTUAL_ENV environment variable
          (setenv "VIRTUAL_ENV" venv-path)

          ;; Remove PYTHONHOME if it exists
          (setenv "PYTHONHOME" nil)

          (message "Activated UV Python environment at %s" venv-path))

      (error "No UV Python environment found in %s" project-root))))

(defun noonker/create-python-project-environment ()
  "Create a Python virtual environment and set up project configuration files."
  (interactive)
  ;; Query for venv name
  (let* ((project-root (projectile-project-root))
         (dir-locals-path (expand-file-name ".dir-locals.el" project-root))
         (envrc-path (expand-file-name ".envrc" project-root))
         (default-directory project-root))

    ;; Create the virtualenv
    (start-process "uv-initialize-venv" "*uv-initialize-venv*" "uv" "venv")

    ;; Create .dir-locals.el file
    (with-temp-file dir-locals-path
      (insert
       "((nil . ((eval . (uv-activate)))))"
       ))

    ;; Create .envrc file
    (with-temp-file envrc-path
      (insert (format "echo \"Activating %s venv\"
source %s.venv/bin/activate
export BASEPATH=%s
export PYTHONPATH=%s
" project-root project-root project-root project-root)))

    ;; Trust the envrc file
    (start-process "direnv-trust" "*direnv-trust*" "direnv" "allow" ".")

    ;; Trust the .dir-locals.el file
    (put 'eval 'safe-local-variable
         (lambda (val) (equal val '(uv-activate))))

    (message "Python environment '%s' created with supporting files" project-root)))

(defun noonker/python-format-with-black ()

    (let ((buf (current-buffer)))

(defun noonker/maybe-create-python-venv ()
  "Automatically create a UV venv if the project contains .py files but no .venv."
  (when (and buffer-file-name
             (projectile-project-p)
             (string-suffix-p ".py" buffer-file-name)
             (not (file-directory-p
                   (expand-file-name ".venv" (projectile-project-root))))
             (executable-find "uv"))
    (message "Python project without venv detected, creating UV environment...")
    (noonker/create-python-project-environment)))

(defun noonker/rofi--insert-entry (file heading entry)
  "Append ENTRY under HEADING in FILE and save it, returning FILE's buffer.
ENTRY goes at the end of the file if HEADING isn't there."
  (with-current-buffer (find-file-noselect (expand-file-name file org-agenda-basedir))
    (org-with-wide-buffer
     (goto-char (point-min))
     (if (re-search-forward (concat "^\\* " heading) nil t)
         (progn (org-end-of-subtree t)
                (unless (bolp) (insert "\n"))
                (insert entry))
       (goto-char (point-max))
       (insert entry)))
    (save-buffer)
    (current-buffer)))

(defun noonker/rofi-capture (text template)
  "Non-interactively capture TEXT using org TEMPLATE key.
Supports: t (Today→Tasks), n (Now + clock-in), b (Backlog→Backlog)."
  (let* ((ts (format-time-string "<%Y-%m-%d %a>"))
         (entry (concat "\n** TODO " text "\n   SCHEDULED: " ts "\n")))
    (pcase template
      ("t" (noonker/rofi--insert-entry "today.org" "Tasks" entry))
      ("n" (with-current-buffer (noonker/rofi--insert-entry "today.org" "Tasks" entry)
             (org-with-wide-buffer
              (goto-char (point-max))
              (when (re-search-backward (regexp-quote (concat "** TODO " text)) nil t)
                (org-clock-in)))))
      ("b" (noonker/rofi--insert-entry "backlog.org" "Backlog" entry))
      (_ (error "noonker/rofi-capture: unknown template key: %s" template)))))

(defun noonker/hex-decode-region (start end)
  "Hex decode the region between START and END.
E.g. 68656c6c6f -> hello"
  (interactive "r")
  (let* ((hex-string (buffer-substring-no-properties start end))
         ;; Remove any whitespace/newlines
         (hex-string (replace-regexp-in-string "[ \t\n\r]+" "" hex-string))
         (len (length hex-string))
         bytes)
    (when (oddp len)
      (error "Hex string has odd length"))
    (when (string-match-p "[^0-9a-fA-F]" hex-string)
      (error "Invalid hex characters in region"))
    (setq bytes (apply #'unibyte-string
                       (cl-loop for i from 0 below len by 2
                                collect (string-to-number
                                         (substring hex-string i (+ i 2))
                                         16))))
    (delete-region start end)
    (insert (decode-coding-string bytes 'raw-text))))

                  (format " *elfeed-playlist-vlc: %s*"

(defun elfeed-playlist--send-norns (files)
  "Upload FILES to norns dust/audio/ via sftp in a date-stamped folder."
  (let* ((folder (downcase (format-time-string "%b%d")))
         (remote-dir (format "dust/audio/%s" folder))
         (host elfeed-playlist-norns-host)
         (user elfeed-playlist-norns-user)
         (buf (generate-new-buffer " *elfeed-playlist-norns*"))
         (cmds (concat (format "mkdir %s\n" remote-dir)
                       (mapconcat (lambda (f)
                                    (format "put %s %s/" f remote-dir))
                                  files "\n")
                       "\nbye\n")))
    (with-current-buffer buf
      (insert cmds))
    (let ((proc (start-process "elfeed-playlist-norns" buf
                               "sftp" "-b" "-"
                               (format "%s@%s" user host))))
      (set-process-query-on-exit-flag proc nil)
      (process-send-string proc cmds)
      (process-send-eof proc)
      (set-process-sentinel
       proc
       (lambda (p _event)
         (when (memq (process-status p) '(exit signal))
           (if (zerop (process-exit-status p))
               (message "elfeed-playlist: uploaded %d file(s) to norns:%s"
                        (length files) remote-dir)

                      (process-exit-status p)

(defun elfeed-playlist--collect-urls ()
  "Return a list of URLs from all entries in the current elfeed search."
  (unless (derived-mode-p 'elfeed-search-mode)
    (user-error "Not in an elfeed-search buffer"))
  (cl-loop for entry in elfeed-search-entries
           for link = (elfeed-entry-link entry)
           when link collect link))

(defun elfeed-playlist--download (urls callback)
  "Download audio for URLS with yt-dlp, then call CALLBACK with the file list.
Runs at most `elfeed-playlist-max-concurrent' downloads at a time."
  (let* ((dir elfeed-playlist-download-dir)
         (state (list :queue (copy-sequence urls)
                      :active 0
                      :total (length urls)
                      :files nil
                      :callback callback
                      :dir dir)))
    (make-directory dir t)
    (message "elfeed-playlist: downloading %d entries..." (length urls))
    (elfeed-playlist--mode-line-update state)
    (elfeed-playlist--drain state)))

(defun elfeed-playlist--drain (state)
  "Start downloads from STATE's queue up to the concurrency limit."
  (while (and (plist-get state :queue)
              (< (plist-get state :active) elfeed-playlist-max-concurrent))
    (let ((url (pop (plist-get state :queue))))
      ;; Mutation: we popped from :queue above, but plist-put is needed
      ;; only for :active since pop already modified the cons in place.
      (plist-put state :active (1+ (plist-get state :active)))
      (elfeed-playlist--start-one url state))))

(defun elfeed-playlist--start-one (url state)
  "Start a single yt-dlp download for URL, updating STATE on completion."
  (let* ((dir (plist-get state :dir))
         (args (append elfeed-playlist-ytdlp-args (list url)))
         (buf (generate-new-buffer " *elfeed-playlist-dl*"))
         (proc (with-current-buffer buf
                 (setq default-directory dir)
                 (apply #'start-process "elfeed-playlist-dl" buf
                        "yt-dlp" args))))
    (set-process-query-on-exit-flag proc nil)
    (set-process-sentinel
     proc
     (lambda (p _event)
       (when (memq (process-status p) '(exit signal))
         (with-current-buffer (process-buffer p)
           (goto-char (point-min))
           (when (re-search-forward
                  "\\[ExtractAudio\\] Destination: \\(.+\\)$" nil t)
             (let ((file (expand-file-name (match-string 1) dir)))
               (when (file-exists-p file)
                 (plist-put state :files
                            (cons file (plist-get state :files))))))
           (goto-char (point-min))
           (when (re-search-forward
                  "\\[download\\] \\(.+\\) has already been downloaded" nil t)
             (let ((file (expand-file-name (match-string 1) dir)))
               (when (file-exists-p file)
                 (plist-put state :files
                            (cons file (plist-get state :files)))))))
         (kill-buffer (process-buffer p))
         (plist-put state :active (1- (plist-get state :active)))
         (elfeed-playlist--mode-line-update state)
         (let ((done (- (plist-get state :total)
                        (plist-get state :active)
                        (length (plist-get state :queue)))))
           (message "elfeed-playlist: %d/%d done"
                    done (plist-get state :total)))
         (if (and (null (plist-get state :queue))
                  (zerop (plist-get state :active)))
             (let ((files (nreverse (plist-get state :files))))
               (elfeed-playlist--mode-line-clear)
               (message "elfeed-playlist: downloads complete (%d files)"
                        (length files))
               (funcall (plist-get state :callback) files))
           (elfeed-playlist--drain state)))))))

(defun elfeed-playlist-download-and-send ()
  "Download audio for all entries in the current elfeed search, then
prompt for a destination."
  (interactive)
  (let ((urls (elfeed-playlist--collect-urls)))
    (when (null urls)
      (user-error "No entries in the current search"))
    (elfeed-playlist--download
     urls
     (lambda (files)
       (if (null files)
           (message "elfeed-playlist: no files were downloaded")
         (let* ((names (mapcar #'car elfeed-playlist-destinations))
                (choice (completing-read
                         (format "Send %d file(s) to: " (length files))
                         names nil t))
                (fn (alist-get choice elfeed-playlist-destinations
                               nil nil #'string=)))
           (funcall fn files)))))))

  "Return the bounds to rewrite: the region, the sentence, or the line."
  (cond
   ((use-region-p) (cons (region-beginning) (region-end)))
   ((derived-mode-p 'text-mode) (bounds-of-thing-at-point 'sentence))
   (t (cons (line-beginning-position) (line-end-position)))))

(defun noonker/gptel--rewrite (bounds system)
  "Replace the text in BOUNDS with a rewrite following SYSTEM."
  (gptel-request
   (buffer-substring-no-properties (car bounds) (cdr bounds))
   :system system
   :buffer (current-buffer)
   :context (cons (set-marker (make-marker) (car bounds))
                  (set-marker (make-marker) (cdr bounds)))
   :callback
   (lambda (response info)
     (if (not response)
         (message "ChatGPT response failed with: %s" (plist-get info :status))
       (let* ((bounds (plist-get info :context))
              (beg (car bounds))
              (end (cdr bounds))
              (buf (plist-get info :buffer)))
         (with-current-buffer buf
           (save-excursion
             (goto-char beg)
             (kill-region beg end)
             (insert response)
             (set-marker beg nil)
             (set-marker end nil)
             (message "Rewrote line. Original line saved to kill-ring."))))))))

    ;; ship MarkupContent with no :value field, which crashes
    ;; eglot--format-markup on signatureHelp responses. Coerce nil
    ;; :value to "" so eldoc doesn't error out.
    (define-advice eglot--format-markup
        (:filter-args (args) sclang-nil-value-fix)
        "Give markup with no :value an empty one, so Eglot doesn't crash on it."
      (let ((markup (car args)))
        (when (and (listp markup) (not (stringp markup))
                   (null (plist-get markup :value)))
          (setcar args (plist-put (copy-sequence markup) :value ""))))
      args))
  (add-hook 'sclang-mode-hook #'eglot-ensure)
  ;; Attach the tree-sitter parser to every sclang-mode buffer so
  ;; `treesit-explore-mode' and other treesit-driven features work.
  ;; The grammar library is `libtree-sitter-supercollider.so'.
  (add-hook 'sclang-mode-hook
            (lambda ()
              (when (treesit-ready-p 'supercollider)
                (treesit-parser-create 'supercollider))))
  )

(defun noonker/norns-rsync-repo ()
  "Async rsync of tracked + untracked (non-ignored) files in the current repo to n:dust/code/<repo>/."
  (interactive)
  (let ((root (locate-dominating-file default-directory ".git")))
    (unless root (user-error "Not in a git repository"))
    (let* ((root (file-name-as-directory (expand-file-name root)))
           (repo (file-name-nondirectory (directory-file-name root)))
           (dest (format "n:dust/code/%s/" repo))
           (default-directory root)
           (buf (get-buffer-create "*norns-rsync*"))
           (cmd (format
                 "git ls-files -c -o --exclude-standard -z | rsync -av --from0 --files-from=- ./ %s"
                 (shell-quote-argument dest))))
      (with-current-buffer buf
        (erase-buffer)
        (insert (format "$ %s\n\n" cmd)))
      (let ((proc (start-process-shell-command "norns-rsync" buf cmd)))
        (process-put proc 'norns-rsync-dest dest)
        (set-process-sentinel
         proc
         (lambda (p _e)
           (when (memq (process-status p) '(exit signal))
             (let ((d (process-get p 'norns-rsync-dest)))
               (if (zerop (process-exit-status p))
                   (message "norns-rsync: %s ✓" d)
                 (message "norns-rsync: failed (exit %d) — see %s"
                          (process-exit-status p)
                          (buffer-name (process-buffer p))))))))))))
